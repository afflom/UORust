definition{
  name: textCode,
  parameters: [bytes : Bytes, offset : Nat],
  result: Nat,
  body: (match (prim index bytes offset : (Option UInt8))
    | Option.none => 256
    | Option.some byte => (prim to_nat byte : Nat)),
  axioms: [Quot.sound, propext]
}

definition{
  name: textCut,
  parameters: [bytes : Bytes, start : Nat, stop : Nat],
  result: String,
  body: (match (prim slice bytes start (prim subtract stop start : Nat) : (Option Bytes))
    | Option.none => ""
    | Option.some piece => (match (prim utf8_decode piece : (Option String))
      | Option.none => ""
      | Option.some text => text)),
  axioms: [Classical.choice, Quot.sound, propext]
}
