definition{

definition{
  name: pow2,
  parameters: [bits : Nat],
  result: Nat,
  body: (match bits | Nat.zero => 1 | Nat.succ rest => (prim multiply 2 (pow2 rest) : Nat)),
  recursive_argument: bits
}

definition{
  name: wrapBits,
  parameters: [bits : Nat, value : Nat],
  result: Nat,
  body: (prim remainder value (pow2 bits) 0 : Nat)
}

definition{
  name: signedOf,
  parameters: [bits : Nat, signed : Bool, repr : Nat],
  result: Signed,
  body: (if (signed && ((pow2 (prim subtract bits 1 : Nat)) <=? repr)) then (record Signed {negative := true, magnitude := (prim subtract (pow2 bits) repr : Nat)}) else (record Signed {negative := false, magnitude := repr}))
}

definition{
  name: reprOf,
  parameters: [bits : Nat, value : Signed],
  result: Nat,
  body: (if ((project value negative) && (not ((wrapBits bits (project value magnitude)) =? 0))) then (prim subtract (pow2 bits) (wrapBits bits (project value magnitude)) : Nat) else (wrapBits bits (project value magnitude)))
}

definition{
  name: fitsIn,
  parameters: [bits : Nat, signed : Bool, value : Signed],
  result: Bool,
  body: (if signed then (if (project value negative) then ((project value magnitude) <=? (pow2 (prim subtract bits 1 : Nat))) else (((project value magnitude) + 1) <=? (pow2 (prim subtract bits 1 : Nat)))) else (((not (project value negative)) || ((project value magnitude) =? 0)) && (((project value magnitude) + 1) <=? (pow2 bits))))
}

definition{
  name: signedLess,
  parameters: [a : Signed, b : Signed],
  result: Bool,
  body: (if (project a negative) then (if (project b negative) then (((project b magnitude) + 1) <=? (project a magnitude)) else (not (((project a magnitude) =? 0) && ((project b magnitude) =? 0)))) else (if (project b negative) then false else (((project a magnitude) + 1) <=? (project b magnitude))))
}

definition{
  name: signedEqual,
  parameters: [a : Signed, b : Signed],
  result: Bool,
  body: (((project a magnitude) =? (project b magnitude)) && ((sameBool (project a negative) (project b negative)) || ((project a magnitude) =? 0)))
}

definition{
  name: signedAdd,
  parameters: [a : Signed, b : Signed],
  result: Signed,
  body: (if (sameBool (project a negative) (project b negative)) then (record Signed {negative := (project a negative), magnitude := ((project a magnitude) + (project b magnitude))}) else (if ((project b magnitude) <=? (project a magnitude)) then (record Signed {negative := (project a negative), magnitude := (prim subtract (project a magnitude) (project b magnitude) : Nat)}) else (record Signed {negative := (project b negative), magnitude := (prim subtract (project b magnitude) (project a magnitude) : Nat)})))
}

definition{
  name: signedNegate,
  parameters: [a : Signed],
  result: Signed,
  body: (record Signed {negative := (not (project a negative)), magnitude := (project a magnitude)})
}

definition{
  name: signedMultiply,
  parameters: [a : Signed, b : Signed],
  result: Signed,
  body: (record Signed {negative := (not (sameBool (project a negative) (project b negative))), magnitude := (prim multiply (project a magnitude) (project b magnitude) : Nat)})
}

definition{
  name: signedQuotient,
  parameters: [a : Signed, b : Signed],
  result: Signed,
  body: (record Signed {negative := (not (sameBool (project a negative) (project b negative))), magnitude := (prim quotient (project a magnitude) (project b magnitude) 0 : Nat)})
}

definition{
  name: signedRemainder,
  parameters: [a : Signed, b : Signed],
  result: Signed,
  body: (record Signed {negative := (project a negative), magnitude := (prim remainder (project a magnitude) (project b magnitude) 0 : Nat)})
}

definition{
  name: bitsAnd,
  parameters: [width : Nat, a : Nat, b : Nat],
  result: Nat,
  body: (match width
    | Nat.zero => 0
    | Nat.succ rest => ((if (((prim remainder a 2 0 : Nat) =? 1) && ((prim remainder b 2 0 : Nat) =? 1)) then 1 else 0) + (prim multiply 2 (bitsAnd rest (prim quotient a 2 0 : Nat) (prim quotient b 2 0 : Nat)) : Nat))),
  recursive_argument: width
}

definition{
  name: bitsOr,
  parameters: [width : Nat, a : Nat, b : Nat],
  result: Nat,
  body: (match width
    | Nat.zero => 0
    | Nat.succ rest => ((if (((prim remainder a 2 0 : Nat) =? 1) || ((prim remainder b 2 0 : Nat) =? 1)) then 1 else 0) + (prim multiply 2 (bitsOr rest (prim quotient a 2 0 : Nat) (prim quotient b 2 0 : Nat)) : Nat))),
  recursive_argument: width
}

definition{
  name: bitsXor,
  parameters: [width : Nat, a : Nat, b : Nat],
  result: Nat,
  body: (match width
    | Nat.zero => 0
    | Nat.succ rest => ((if ((prim remainder a 2 0 : Nat) =? (prim remainder b 2 0 : Nat)) then 0 else 1) + (prim multiply 2 (bitsXor rest (prim quotient a 2 0 : Nat) (prim quotient b 2 0 : Nat)) : Nat))),
  recursive_argument: width
}

definition{
  name: shiftRight,
  parameters: [bits : Nat, signed : Bool, repr : Nat, amount : Nat],
  result: Nat,
  body: (let value : Signed := (signedOf bits signed repr); (if (project value negative) then (reprOf bits (record Signed {negative := true, magnitude := (prim quotient ((project value magnitude) + (prim subtract (pow2 amount) 1 : Nat)) (pow2 amount) 0 : Nat)})) else (prim quotient repr (pow2 amount) 0 : Nat)))
}

definition{
  name: scalarOf,
  parameters: [value : Value],
  result: (Option Signed),
  body: (match value
    | Value.int bits signed repr => (ctor Option.some [Signed] (signedOf bits signed repr))
    | Value.boolean b => (ctor Option.some [Signed] (record Signed {negative := false, magnitude := (if b then 1 else 0)}))
    | Value.character code => (ctor Option.some [Signed] (record Signed {negative := false, magnitude := code}))
    | Value.unit => (ctor Option.none [Signed])
    | Value.tuple fields => (ctor Option.none [Signed])
    | Value.adt variant discriminant fields => (ctor Option.none [Signed])
    | Value.reference steps => (ctor Option.none [Signed])
    | Value.pointer frame base steps => (ctor Option.none [Signed])
    | Value.function path => (ctor Option.none [Signed])
    | Value.text content => (ctor Option.none [Signed])
    | Value.uninit => (ctor Option.none [Signed]))
}

definition{
  name: orderingOf,
  parameters: [a : Signed, b : Signed],
  result: Value,
  body: (if (signedLess a b) then (ctor Value.adt "Less" (record Signed {negative := true, magnitude := 1}) (nil Value)) else (if (signedEqual a b) then (ctor Value.adt "Equal" (record Signed {negative := false, magnitude := 0}) (nil Value)) else (ctor Value.adt "Greater" (record Signed {negative := false, magnitude := 1}) (nil Value))))
}

definition{
  name: evalCompare,
  parameters: [op : BinOp, a : Signed, b : Signed],
  result: Evaluated,
  body: (match op
    | BinOp.add => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.addUnchecked => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.addWithOverflow => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.sub => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.subUnchecked => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.subWithOverflow => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.mul => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.mulUnchecked => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.mulWithOverflow => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.div => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.rem => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.bitXor => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.bitAnd => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.bitOr => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.shl => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.shlUnchecked => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.shr => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.shrUnchecked => (ctor Evaluated.stop (ctor Outcome.ub "an arithmetic operator on values that are not integers"))
    | BinOp.eq => (boolValue (signedEqual a b))
    | BinOp.lt => (boolValue (signedLess a b))
    | BinOp.le => (boolValue (not (signedLess b a)))
    | BinOp.ne => (boolValue (not (signedEqual a b)))
    | BinOp.ge => (boolValue (not (signedLess a b)))
    | BinOp.gt => (boolValue (signedLess b a))
    | BinOp.cmp => (ctor Evaluated.found (orderingOf a b))
    | BinOp.offset => (ctor Evaluated.stop (ctor Outcome.undefined "BinOp::Offset")))
}

definition{
  name: exactValue,
  parameters: [bits : Nat, signed : Bool, value : Signed],
  result: Evaluated,
  body: (if (fitsIn bits signed value) then (ctor Evaluated.found (intValue bits signed value)) else (ctor Evaluated.stop (ctor Outcome.ub "an unchecked operation overflowed")))
}

definition{
  name: flaggedValue,
  parameters: [bits : Nat, signed : Bool, value : Signed],
  result: Evaluated,
  body: (ctor Evaluated.found (ctor Value.tuple (cons (intValue bits signed value) (cons (ctor Value.boolean (not (fitsIn bits signed value))) (nil Value)))))
}

definition{
  name: evalDivide,
  parameters: [bits : Nat, signed : Bool, a : Signed, b : Signed, remainder : Bool],
  result: Evaluated,
  body: (if ((project b magnitude) =? 0) then (ctor Evaluated.stop (ctor Outcome.ub "a division by zero")) else (let q : Signed := (signedQuotient a b); (if (fitsIn bits signed q) then (ctor Evaluated.found (intValue bits signed (if remainder then (signedRemainder a b) else q))) else (ctor Evaluated.stop (ctor Outcome.ub "a division that overflows")))))
}

definition{
  name: evalArithmetic,
  parameters: [op : BinOp, bits : Nat, signed : Bool, a : Nat, b : Nat, amount : Signed],
  result: Evaluated,
  body: (let x : Signed := (signedOf bits signed a); (let y : Signed := (signedOf bits signed b); (match op
    | BinOp.add => (ctor Evaluated.found (intValue bits signed (signedAdd x y)))
    | BinOp.addUnchecked => (exactValue bits signed (signedAdd x y))
    | BinOp.addWithOverflow => (flaggedValue bits signed (signedAdd x y))
    | BinOp.sub => (ctor Evaluated.found (intValue bits signed (signedAdd x (signedNegate y))))
    | BinOp.subUnchecked => (exactValue bits signed (signedAdd x (signedNegate y)))
    | BinOp.subWithOverflow => (flaggedValue bits signed (signedAdd x (signedNegate y)))
    | BinOp.mul => (ctor Evaluated.found (intValue bits signed (signedMultiply x y)))
    | BinOp.mulUnchecked => (exactValue bits signed (signedMultiply x y))
    | BinOp.mulWithOverflow => (flaggedValue bits signed (signedMultiply x y))
    | BinOp.div => (evalDivide bits signed x y false)
    | BinOp.rem => (evalDivide bits signed x y true)
    | BinOp.bitXor => (ctor Evaluated.found (ctor Value.int bits signed (bitsXor bits a b)))
    | BinOp.bitAnd => (ctor Evaluated.found (ctor Value.int bits signed (bitsAnd bits a b)))
    | BinOp.bitOr => (ctor Evaluated.found (ctor Value.int bits signed (bitsOr bits a b)))
    | BinOp.shl => (ctor Evaluated.found (ctor Value.int bits signed (wrapBits bits (prim multiply a (pow2 (prim remainder (reprOf bits amount) bits 0 : Nat)) : Nat))))
    | BinOp.shlUnchecked => (if ((not (project amount negative)) && (((project amount magnitude) + 1) <=? bits)) then (ctor Evaluated.found (ctor Value.int bits signed (wrapBits bits (prim multiply a (pow2 (project amount magnitude)) : Nat)))) else (ctor Evaluated.stop (ctor Outcome.ub "an unchecked shift by the width or more")))
    | BinOp.shr => (ctor Evaluated.found (ctor Value.int bits signed (shiftRight bits signed a (prim remainder (reprOf bits amount) bits 0 : Nat))))

    | BinOp.eq => (evalCompare op x y)
    | BinOp.lt => (evalCompare op x y)
    | BinOp.le => (evalCompare op x y)
    | BinOp.ne => (evalCompare op x y)
    | BinOp.ge => (evalCompare op x y)
    | BinOp.gt => (evalCompare op x y)
    | BinOp.cmp => (evalCompare op x y)
    | BinOp.offset => (ctor Evaluated.stop (ctor Outcome.undefined "BinOp::Offset")))))
}

    | Value.adt variant discriminant old => (ctor Value.adt variant discriminant fields)
    | Value.tuple old => (ctor Value.tuple fields)
    | Value.int p q r => (ctor Value.tuple fields)
    | Value.boolean p => (ctor Value.tuple fields)
    | Value.character p => (ctor Value.tuple fields)
    | Value.unit => (ctor Value.tuple fields)
    | Value.reference r => (ctor Value.tuple fields)
    | Value.pointer p q r => (ctor Value.tuple fields)
    | Value.function p => (ctor Value.tuple fields)
    | Value.text p => (ctor Value.tuple fields)
    | Value.uninit => (ctor Value.tuple fields))
}

definition{
  name: readSteps,
  parameters: [steps : (List Nat), value : Value],
  result: (Option Value),
  body: (match steps
    | List.nil => (ctor Option.some [Value] value)
    | List.cons step rest => (match (fieldsOf value)
      | Option.none => (ctor Option.none [Value])
      | Option.some fields => (match (nthValue fields step)
        | Option.none => (ctor Option.none [Value])
        | Option.some inner => (readSteps rest inner)))),
  recursive_argument: steps
}

definition{
  name: writeSteps,
  parameters: [steps : (List Nat), value : Value, replacement : Value],
  result: (Option Value),
  body: (match steps
    | List.nil => (ctor Option.some [Value] replacement)
    | List.cons step rest => (match (fieldsOf value)
      | Option.none => (match value
        | Value.uninit => (match (writeSteps rest (ctor Value.uninit) replacement)
          | Option.none => (ctor Option.none [Value])
          | Option.some inner => (ctor Option.some [Value] (ctor Value.tuple (setValue (uninits (step + 1)) step inner))))
        | Value.int p q r => (ctor Option.none [Value])
        | Value.boolean p => (ctor Option.none [Value])
        | Value.character p => (ctor Option.none [Value])
        | Value.unit => (ctor Option.none [Value])
        | Value.tuple p => (ctor Option.none [Value])
        | Value.adt p q r => (ctor Option.none [Value])
        | Value.reference r => (ctor Option.none [Value])
        | Value.pointer p q r => (ctor Option.none [Value])
        | Value.function p => (ctor Option.none [Value])
        | Value.text p => (ctor Option.none [Value]))
      | Option.some fields => (let padded : (List Value) := (if ((step + 1) <=? (countValues fields)) then fields else (appendValues fields (uninits (prim subtract (step + 1) (countValues fields) : Nat)))); (match (nthValue padded step)
        | Option.none => (ctor Option.none [Value])
        | Option.some inner => (match (writeSteps rest inner replacement)
          | Option.none => (ctor Option.none [Value])
          | Option.some updated => (ctor Option.some [Value] (withFields value (setValue padded step updated)))))))),
  recursive_argument: steps
}

definition{
  name: countFrames,
  parameters: [frames : (List Frame)],
  result: Nat,
  body: (match frames | List.nil => 0 | List.cons head rest => ((countFrames rest) + 1)),
  recursive_argument: frames
}

  parameters: [frames : (List Frame), depth : Nat],
  result: (Option Frame),
  body: (match frames
    | List.nil => (ctor Option.none [Frame])
    | List.cons head rest => (if ((countFrames rest) =? depth) then (ctor Option.some [Frame] head) else (frameAt rest depth))),
  recursive_argument: frames
}

definition{

    | Option.none => (match (frameAt frames (project loc frame))
      | Option.none => (ctor Option.none [Value])
      | Option.some frame => (match (nthValue (project frame locals) (project loc base))
        | Option.none => (ctor Option.none [Value])

    | Option.some frame => (match (nthValue (project frame locals) (project loc base))

      | Option.some whole => (match (writeSteps (project loc steps) whole value)
        | Option.none => (ctor Option.none [(List Frame)])
        | Option.some updated => (ctor Option.some [(List Frame)] (setFrame frames (project loc frame) (project loc base) updated))))))
}

definition{
  name: locateSteps,
  parameters: [projection : (List Proj), frames : (List Frame), here : Nat, loc : Location],
  result: Located,
  body: (match projection
    | List.nil => (ctor Located.found loc)
    | List.cons step rest => (match step
      | Proj.deref => (match (readAt frames loc)
        | Option.some value => (match value
          | Value.pointer frame base steps => (locateSteps rest frames here (locationOf frame base steps))
          | Value.reference target => (locateSteps rest frames here (record Location {frame := 0, base := 0, steps := (nil Nat), root := (ctor Option.some [Value] target)}))
          | Value.int p q r => (ctor Located.stop (ctor Outcome.undefined "ProjectionElem::Deref of a value that is not a reference"))
          | Value.boolean p => (ctor Located.stop (ctor Outcome.ub "a dereference of a value that is not a pointer"))
          | Value.character p => (ctor Located.stop (ctor Outcome.ub "a dereference of a value that is not a pointer"))
          | Value.unit => (ctor Located.stop (ctor Outcome.ub "a dereference of a value that is not a pointer"))
          | Value.tuple p => (ctor Located.stop (ctor Outcome.undefined "ProjectionElem::Deref of a value that is not a reference"))
          | Value.adt p q r => (ctor Located.stop (ctor Outcome.undefined "ProjectionElem::Deref of a value that is not a reference"))
          | Value.function p => (ctor Located.stop (ctor Outcome.ub "a dereference of a value that is not a pointer"))
          | Value.text p => (ctor Located.stop (ctor Outcome.undefined "ProjectionElem::Deref of a value that is not a reference"))
          | Value.uninit => (ctor Located.stop (ctor Outcome.ub "a dereference of an uninitialized pointer")))
        | Option.none => (ctor Located.stop (ctor Outcome.ub "a dereference of a place that holds no value")))
      | Proj.field index => (locateSteps rest frames here (extendAt loc index))
      | Proj.index base => (match (readAt frames (locationOf here base (nil Nat)))
        | Option.some value => (match (scalarOf value)
          | Option.some offset => (locateSteps rest frames here (extendAt loc (project offset magnitude)))
          | Option.none => (ctor Located.stop (ctor Outcome.ub "an index that is not an integer")))
        | Option.none => (ctor Located.stop (ctor Outcome.ub "an index that holds no value")))
      | Proj.constantIndex offset fromEnd => (if fromEnd then (match (readAt frames loc)
        | Option.some value => (match (fieldsOf value)
          | Option.some fields => (locateSteps rest frames here (extendAt loc (prim subtract (countValues fields) offset : Nat)))
          | Option.none => (ctor Located.stop (ctor Outcome.ub "a constant index into a value that is not an array")))
        | Option.none => (ctor Located.stop (ctor Outcome.ub "a constant index into a place that holds no value"))) else (locateSteps rest frames here (extendAt loc offset)))
      | Proj.downcast variant => (locateSteps rest frames here loc)
      | Proj.subslice start finish fromEnd => (match (readAt frames loc)
        | Option.some value => (match (fieldsOf value)
          | Option.some fields => (locateSteps rest frames here (pinnedAt (ctor Value.tuple (takeValues (prim subtract (if fromEnd then (prim subtract (countValues fields) finish : Nat) else finish) start : Nat) (skipValues start fields)))))
          | Option.none => (ctor Located.stop (ctor Outcome.ub "a subslice of a value that is not an array")))
        | Option.none => (ctor Located.stop (ctor Outcome.ub "a subslice of a place that holds no value")))
      | Proj.other kind text => (ctor Located.stop (ctor Outcome.undefined kind)))),
  recursive_argument: projection
}

definition{
  name: evalRvalue,
  parameters: [program : Program, frames : (List Frame), value : Rvalue],
  result: Evaluated,
  body: (match value
    | Rvalue.value op => (evalOperand program frames op)
    | Rvalue.borrow kind source => (match (placeOf frames source)
      | Located.stop outcome => (ctor Evaluated.stop outcome)
      | Located.found loc => (pointerTo loc))
    | Rvalue.copyForDeref source => (readPlace frames source)
    | Rvalue.binary op left right => (match (evalOperand program frames left)
      | Evaluated.stop outcome => (ctor Evaluated.stop outcome)
      | Evaluated.found a => (match (evalOperand program frames right)
        | Evaluated.stop outcome => (ctor Evaluated.stop outcome)
        | Evaluated.found b => (evalBinary op a b)))
    | Rvalue.unary op inner => (match (evalOperand program frames inner)
      | Evaluated.stop outcome => (ctor Evaluated.stop outcome)
      | Evaluated.found a => (match op
        | UnOp.ptrMetadata => (metadataOf program frames a)
        | UnOp.not => (evalUnary op a)
        | UnOp.neg => (evalUnary op a)))
    | Rvalue.discriminant source target => (match (readPlace frames source)
      | Evaluated.stop outcome => (ctor Evaluated.stop outcome)
      | Evaluated.found whole => (match whole
        | Value.adt variant discriminant fields => (match target
          | Ty.int bits signed => (ctor Evaluated.found (intValue bits signed discriminant))
          | Ty.boolean => (ctor Evaluated.stop (ctor Outcome.ub "a discriminant read at a type that is not an integer"))
          | Ty.character => (ctor Evaluated.stop (ctor Outcome.ub "a discriminant read at a type that is not an integer"))
          | Ty.unit => (ctor Evaluated.stop (ctor Outcome.ub "a discriminant read at a type that is not an integer"))
          | Ty.float e m => (ctor Evaluated.stop (ctor Outcome.ub "a discriminant read at a type that is not an integer"))
          | Ty.other printed => (ctor Evaluated.stop (ctor Outcome.undefined "Rvalue::Discriminant at a type the embedding does not read")))
        | Value.int p q r => (ctor Evaluated.stop (ctor Outcome.undefined "Rvalue::Discriminant of a value that is not an enum"))
        | Value.boolean p => (ctor Evaluated.stop (ctor Outcome.undefined "Rvalue::Discriminant of a value that is not an enum"))
        | Value.character p => (ctor Evaluated.stop (ctor Outcome.undefined "Rvalue::Discriminant of a value that is not an enum"))
        | Value.unit => (ctor Evaluated.stop (ctor Outcome.undefined "Rvalue::Discriminant of a value that is not an enum"))
        | Value.tuple p => (ctor Evaluated.stop (ctor Outcome.undefined "Rvalue::Discriminant of a value that is not an enum"))
        | Value.reference r => (ctor Evaluated.stop (ctor Outcome.undefined "Rvalue::Discriminant of a value that is not an enum"))
        | Value.pointer p q r => (ctor Evaluated.stop (ctor Outcome.undefined "Rvalue::Discriminant of a value that is not an enum"))
        | Value.function p => (ctor Evaluated.stop (ctor Outcome.undefined "Rvalue::Discriminant of a value that is not an enum"))
        | Value.text p => (ctor Evaluated.stop (ctor Outcome.undefined "Rvalue::Discriminant of a value that is not an enum"))
        | Value.uninit => (ctor Evaluated.stop (ctor Outcome.ub "a read of an uninitialized value"))))
    | Rvalue.aggregate kind ops => (match (evalOperands program frames ops)
      | EvaluatedList.stop outcome => (ctor Evaluated.stop outcome)
      | EvaluatedList.found values => (evalAggregate kind values))
    | Rvalue.cast kind inner target => (match (evalOperand program frames inner)
      | Evaluated.stop outcome => (ctor Evaluated.stop outcome)
      | Evaluated.found a => (evalCast kind a target))
    | Rvalue.floatBinary e m op left right => (match (evalOperand program frames left)
      | Evaluated.stop outcome => (ctor Evaluated.stop outcome)
      | Evaluated.found a => (match (evalOperand program frames right)
        | Evaluated.stop outcome => (ctor Evaluated.stop outcome)
        | Evaluated.found b => (match (floatBitsOf a)
          | Option.none => (ctor Evaluated.stop (ctor Outcome.ub "a float operation on a value that is not a float"))
          | Option.some x => (match (floatBitsOf b)
            | Option.none => (ctor Evaluated.stop (ctor Outcome.ub "a float operation on a value that is not a float"))
            | Option.some y => (floatArith e m op x y)))))
    | Rvalue.floatNeg e m inner => (match (evalOperand program frames inner)
      | Evaluated.stop outcome => (ctor Evaluated.stop outcome)
      | Evaluated.found a => (match (floatBitsOf a)
        | Option.none => (ctor Evaluated.stop (ctor Outcome.ub "a float operation on a value that is not a float"))
        | Option.some x => (ctor Evaluated.found (floatValue e m (Prelude::floatNeg e m x)))))
    | Rvalue.numeric source target inner => (match (evalOperand program frames inner)
      | Evaluated.stop outcome => (ctor Evaluated.stop outcome)
      | Evaluated.found a => (evalNumeric source target a))
    | Rvalue.rawPtr source => (match (placeOf frames source)
      | Located.stop outcome => (ctor Evaluated.stop outcome)
      | Located.found loc => (pointerTo loc))
    | Rvalue.repeat inner generic => (match (genericCount frames generic)
      | Option.none => (ctor Evaluated.stop (ctor Outcome.undefined "a generic argument whose count its caller does not supply"))
      | Option.some count => (match (evalOperand program frames inner)
        | Evaluated.stop outcome => (ctor Evaluated.stop outcome)
        | Evaluated.found a => (ctor Evaluated.found (ctor Value.tuple (repeatValue count a)))))
    | Rvalue.other kind text => (ctor Evaluated.stop (ctor Outcome.undefined kind)))
}

  name: findBinding,
  parameters: [bindings : (List Binding), name : String],
  result: (Option String),

            | Option.none => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
            | Option.some fields => (if ((countValues fields) =? count) then (encodeWork rest big (appendPieces (elementPieces (project piece offset) stride element fields) work) bytes) else (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))))
          | Value.boolean p => (match (fieldsOf (project piece value))
            | Option.none => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
            | Option.some fields => (if ((countValues fields) =? count) then (encodeWork rest big (appendPieces (elementPieces (project piece offset) stride element fields) work) bytes) else (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))))
          | Value.character p => (match (fieldsOf (project piece value))
            | Option.none => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
            | Option.some fields => (if ((countValues fields) =? count) then (encodeWork rest big (appendPieces (elementPieces (project piece offset) stride element fields) work) bytes) else (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))))
          | Value.unit => (match (fieldsOf (project piece value))
            | Option.none => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
            | Option.some fields => (if ((countValues fields) =? count) then (encodeWork rest big (appendPieces (elementPieces (project piece offset) stride element fields) work) bytes) else (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))))
          | Value.tuple p => (match (fieldsOf (project piece value))
            | Option.none => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
            | Option.some fields => (if ((countValues fields) =? count) then (encodeWork rest big (appendPieces (elementPieces (project piece offset) stride element fields) work) bytes) else (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))))
          | Value.adt p q r => (match (fieldsOf (project piece value))
            | Option.none => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
            | Option.some fields => (if ((countValues fields) =? count) then (encodeWork rest big (appendPieces (elementPieces (project piece offset) stride element fields) work) bytes) else (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))))
          | Value.pointer p q r => (match (fieldsOf (project piece value))
            | Option.none => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
            | Option.some fields => (if ((countValues fields) =? count) then (encodeWork rest big (appendPieces (elementPieces (project piece offset) stride element fields) work) bytes) else (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))))
          | Value.reference p => (match (fieldsOf (project piece value))
            | Option.none => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
            | Option.some fields => (if ((countValues fields) =? count) then (encodeWork rest big (appendPieces (elementPieces (project piece offset) stride element fields) work) bytes) else (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))))
          | Value.function p => (match (fieldsOf (project piece value))
            | Option.none => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
            | Option.some fields => (if ((countValues fields) =? count) then (encodeWork rest big (appendPieces (elementPieces (project piece offset) stride element fields) work) bytes) else (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))))
          | Value.text p => (match (fieldsOf (project piece value))
            | Option.none => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
            | Option.some fields => (if ((countValues fields) =? count) then (encodeWork rest big (appendPieces (elementPieces (project piece offset) stride element fields) work) bytes) else (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))))
          | Value.uninit => (encodeWork rest big work bytes))
        | Layout.enumeration size signed names discriminants => (match (project piece value)
          | Value.int p q r => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
          | Value.boolean p => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
          | Value.character p => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
          | Value.unit => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
          | Value.tuple p => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
          | Value.adt variant discriminant fields => (encodeWork rest big work (spliceBytes bytes (project piece offset) (orderBytes big (littleBytes size (reprOf (prim multiply 8 size : Nat) discriminant)))))
          | Value.pointer p q r => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
          | Value.reference p => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
          | Value.function p => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
          | Value.text p => (ctor Encoded.stop (ctor Outcome.undefined "a transmute of a value its layout does not hold"))
          | Value.uninit => (encodeWork rest big work bytes))
        | Layout.ranged base low high => (encodeWork rest big (cons (record Piece {offset := (project piece offset), layout := base, value := (project piece value)}) work) bytes))))),
  recursive_argument: fuel
}

definition{
  name: fieldTasks,
  parameters: [offset : Nat, offsets : (List Nat), layouts : (List Layout), rest : (List Task)],
  result: (List Task),
  body: (match layouts
    | List.nil => rest
    | List.cons layout others => (match offsets
      | List.nil => rest
      | List.cons at ats => (cons (ctor Task.piece (offset + at) layout) (fieldTasks offset ats others rest)))),
  recursive_argument: layouts
}

definition{
  name: elementTasks,
  parameters: [count : Nat, offset : Nat, stride : Nat, element : Layout, rest : (List Task)],
  result: (List Task),
  body: (match count
    | Nat.zero => rest
    | Nat.succ more => (cons (ctor Task.piece offset element) (elementTasks more (offset + stride) stride element rest))),
  recursive_argument: count
}

definition{

  parameters: [count : Nat, stack : (List Value), acc : (List Value)],
  result: (List Value),
  body: (match count
    | Nat.zero => acc
    | Nat.succ rest => (match stack
      | List.nil => acc
      | List.cons v vs => (takeOnto rest vs (cons v acc)))),
  recursive_argument: count
}

definition{

  parameters: [count : Nat, stack : (List Value)],
  result: (List Value),
  body: (match count
    | Nat.zero => stack
    | Nat.succ rest => (match stack
      | List.nil => (nil Value)
      | List.cons v vs => (dropValues rest vs))),
  recursive_argument: count
}

definition{
  name: variantOf,
  parameters: [names : (List String), discriminants : (List Signed), tag : Signed],
  result: (Option Value),

    | List.nil => (ctor Option.none [Value])
    | List.cons name others => (match discriminants
      | List.nil => (ctor Option.none [Value])
      | List.cons d ds => (if (signedEqual d tag) then (ctor Option.some [Value] (ctor Value.adt name d (nil Value))) else (variantOf others ds tag)))),
  recursive_argument: names
}

definition{
  name: validChar,
  parameters: [code : Nat],
  result: Bool,
  body: ((code <=? 1114111) && (((code + 1) <=? 55296) || (57344 <=? code)))
}

definition{
  name: decodeWork,
  parameters: [fuel : Nat, big : Bool, bytes : (List Byte), work : (List Task), stack : (List Value)],
  result: Evaluated,
  body: (match fuel
    | Nat.zero => (ctor Evaluated.stop (ctor Outcome.exhausted))
    | Nat.succ rest => (match work
      | List.nil => (match stack
        | List.nil => (ctor Evaluated.stop (ctor Outcome.ub "a transmute that decodes no value"))
        | List.cons v vs => (ctor Evaluated.found v))
      | List.cons task tasks => (match task
        | Task.tuple count => (decodeWork rest big bytes tasks (cons (ctor Value.tuple (takeOnto count stack (nil Value))) (dropValues count stack)))
        | Task.adt count => (decodeWork rest big bytes tasks (cons (ctor Value.adt "" (record Signed {negative := false, magnitude := 0}) (takeOnto count stack (nil Value))) (dropValues count stack)))
        | Task.check low high => (match stack
          | List.nil => (ctor Evaluated.stop (ctor Outcome.ub "a transmute that decodes no value"))
          | List.cons v vs => (if (if (low <=? high) then ((low <=? (reprOfValue v)) && ((reprOfValue v) <=? high)) else ((low <=? (reprOfValue v)) || ((reprOfValue v) <=? high))) then (decodeWork rest big bytes tasks stack) else (ctor Evaluated.stop (ctor Outcome.ub "a transmute to a value outside its type's valid range"))))
        | Task.piece offset layout => (let region : (List Byte) := (takeBytes (layoutSize layout) (dropBytes offset bytes)); (match layout
          | Layout.int size signed => (if (hasFragment region) then (ctor Evaluated.stop (ctor Outcome.undefined "a transmute of a pointer's bytes to a scalar")) else (match (littleNumber (orderBytes big region))
            | Option.none => (ctor Evaluated.stop (ctor Outcome.ub "a transmute of uninitialized bytes to a scalar"))
            | Option.some n => (decodeWork rest big bytes tasks (cons (ctor Value.int (prim multiply 8 size : Nat) signed n) stack))))
          | Layout.boolean => (if (hasFragment region) then (ctor Evaluated.stop (ctor Outcome.undefined "a transmute of a pointer's bytes to a scalar")) else (match (littleNumber (orderBytes big region))
            | Option.none => (ctor Evaluated.stop (ctor Outcome.ub "a transmute of uninitialized bytes to a scalar"))
            | Option.some n => (if (n <=? 1) then (decodeWork rest big bytes tasks (cons (ctor Value.boolean (n =? 1)) stack)) else (ctor Evaluated.stop (ctor Outcome.ub "a transmute to a bool of a byte that is neither 0 nor 1")))))
          | Layout.character => (if (hasFragment region) then (ctor Evaluated.stop (ctor Outcome.undefined "a transmute of a pointer's bytes to a scalar")) else (match (littleNumber (orderBytes big region))
            | Option.none => (ctor Evaluated.stop (ctor Outcome.ub "a transmute of uninitialized bytes to a scalar"))
            | Option.some n => (if (validChar n) then (decodeWork rest big bytes tasks (cons (ctor Value.character n) stack)) else (ctor Evaluated.stop (ctor Outcome.ub "a transmute to a char of a value that is not a Unicode scalar value")))))
          | Layout.pointer size => (match region
            | List.nil => (ctor Evaluated.stop (ctor Outcome.ub "a transmute to a pointer of too few bytes"))
            | List.cons lead others => (match lead
              | Byte.fragment v i => (if ((i =? 0) && ((fragmentsFrom 1 others) && ((countBytes region) =? size))) then (decodeWork rest big bytes tasks (cons v stack)) else (ctor Evaluated.stop (ctor Outcome.undefined "a transmute to a pointer of bytes not one pointer's")))
              | Byte.init n => (ctor Evaluated.stop (ctor Outcome.undefined "a transmute to a pointer of bytes not one pointer's"))
              | Byte.uninit => (ctor Evaluated.stop (ctor Outcome.ub "a transmute to a pointer of uninitialized bytes"))))
          | Layout.record size adt offsets layouts => (decodeWork rest big bytes (fieldTasks offset offsets layouts (cons (if adt then (ctor Task.adt (countLayouts layouts)) else (ctor Task.tuple (countLayouts layouts))) tasks)) stack)
          | Layout.array element count stride => (decodeWork rest big bytes (elementTasks count offset stride element (cons (ctor Task.tuple count) tasks)) stack)
          | Layout.enumeration size signed names discriminants => (if (hasFragment region) then (ctor Evaluated.stop (ctor Outcome.undefined "a transmute of a pointer's bytes to a scalar")) else (match (littleNumber (orderBytes big region))
            | Option.none => (ctor Evaluated.stop (ctor Outcome.ub "a transmute of uninitialized bytes to a scalar"))
            | Option.some n => (match (variantOf names discriminants (signedOf (prim multiply 8 size : Nat) signed n)) | Option.none => (ctor Evaluated.stop (ctor Outcome.ub "a transmute to an enum of a discriminant it does not have")) | Option.some v => (decodeWork rest big bytes tasks (cons v stack)))))
          | Layout.ranged base low high => (decodeWork rest big bytes (cons (ctor Task.piece offset base) (cons (ctor Task.check low high) tasks)) stack)))))),
  recursive_argument: fuel
}

definition{
  name: transmuteValue,
  parameters: [fuel : Nat, big : Bool, source : Layout, target2 : Layout, value : Value],
  result: Evaluated,
  body: (if ((layoutSize source) =? (layoutSize target2)) then (match (encodeWork fuel big (cons (record Piece {offset := 0, layout := source, value := value}) (nil Piece)) (uninitBytes (layoutSize source)))
    | Encoded.stop outcome => (ctor Evaluated.stop outcome)
    | Encoded.found bytes => (decodeWork fuel big bytes (cons (ctor Task.piece 0 target2) (nil Task)) (nil Value))) else (ctor Evaluated.stop (ctor Outcome.ub "a transmute between types of different sizes")))
}

definition{
  name: writeRun,
  parameters: [frames : (List Frame), loc : Location, value : Value, count : Nat],
  result: (Option (List Frame)),
  body: (match count
    | Nat.zero => (ctor Option.some [(List Frame)] frames)
    | Nat.succ rest => (match (writeAt frames loc value)
      | Option.none => (ctor Option.none [(List Frame)])
      | Option.some written => (writeRun written (record Location {frame := (project loc frame), base := (project loc base), steps := (bumpLast (project loc steps)), root := (project loc root)}) value rest))),

}

definition{
  name: writeBytes,
  parameters: [fuel : Nat, big : Bool, frames : (List Frame), element : Layout, values : (List Value), destination : Place, target : (Option Nat)],
  result: Outcome,
  body: (match values
    | List.nil => (ctor Outcome.ub "a write_bytes to no pointer")
    | List.cons pointer rest => (match rest
      | List.nil => (ctor Outcome.ub "a write_bytes of no byte")
      | List.cons byte more => (match more
        | List.nil => (ctor Outcome.ub "a write_bytes of no count")
        | List.cons count others => (match (placeOfPointer pointer)
          | Option.none => (ctor Outcome.ub "a write_bytes to a value that is not a pointer to a place")
          | Option.some loc => (match (scalarOf byte)
            | Option.none => (ctor Outcome.ub "a write_bytes of a byte that is not an integer")
            | Option.some b => (match (scalarOf count)
              | Option.none => (ctor Outcome.ub "a write_bytes of a count that is not an integer")
              | Option.some n => (match (transmuteValue fuel big (ctor Layout.array (ctor Layout.int 1 false) (layoutSize element) 1) element (ctor Value.tuple (repeatValue (layoutSize element) (ctor Value.int 8 false (project b magnitude)))))
                | Evaluated.stop outcome => outcome
                | Evaluated.found filled => (match (writeRun frames loc filled (project n magnitude))
                  | Option.none => (ctor Outcome.ub "a write_bytes to a place that holds no value")
                  | Option.some written => (finishCall written destination target (ctor Value.unit))))))))))
}

definition{
  name: pinWork,
  parameters: [fuel : Nat, depth : Nat, frames : (List Frame), work : (List Pin), stack : (List Value)],
  result: Evaluated,
  body: (match fuel
    | Nat.zero => (ctor Evaluated.stop (ctor Outcome.exhausted))
    | Nat.succ rest => (match work
      | List.nil => (match stack
        | List.nil => (ctor Evaluated.stop (ctor Outcome.ub "a constant with no value"))
        | List.cons v vs => (ctor Evaluated.found v))
      | List.cons task tasks => (match task
        | Pin.visit v => (match v
          | Value.int p q r => (pinWork rest depth frames tasks (cons v stack))
          | Value.boolean p => (pinWork rest depth frames tasks (cons v stack))
          | Value.character p => (pinWork rest depth frames tasks (cons v stack))
          | Value.unit => (pinWork rest depth frames tasks (cons v stack))
          | Value.tuple fields => (pinWork rest depth frames (visitTasks fields (cons (ctor Pin.tuple (countValues fields)) tasks)) stack)
          | Value.adt variant discriminant fields => (pinWork rest depth frames (visitTasks fields (cons (ctor Pin.adt variant discriminant (countValues fields)) tasks)) stack)
          | Value.pointer frame base steps => (if (frame =? depth) then (match (readAt frames (locationOf frame base steps))
              | Option.none => (ctor Evaluated.stop (ctor Outcome.ub "a constant's value points to a place that holds no value"))
              | Option.some inner => (pinWork rest depth frames (cons (ctor Pin.visit inner) (cons (ctor Pin.pin) tasks)) stack)) else (pinWork rest depth frames tasks (cons v stack)))
          | Value.reference inner => (pinWork rest depth frames (cons (ctor Pin.visit inner) (cons (ctor Pin.pin) tasks)) stack)
          | Value.function p => (pinWork rest depth frames tasks (cons v stack))
          | Value.text p => (pinWork rest depth frames tasks (cons v stack))
          | Value.uninit => (pinWork rest depth frames tasks (cons v stack)))
        | Pin.tuple count => (pinWork rest depth frames tasks (cons (ctor Value.tuple (takeOnto count stack (nil Value))) (dropValues count stack)))
        | Pin.adt variant discriminant count => (pinWork rest depth frames tasks (cons (ctor Value.adt variant discriminant (takeOnto count stack (nil Value))) (dropValues count stack)))
        | Pin.pin => (match stack
          | List.nil => (ctor Evaluated.stop (ctor Outcome.ub "a constant with no value"))
          | List.cons v vs => (pinWork rest depth frames tasks (cons (ctor Value.reference v) vs)))))),
  recursive_argument: fuel
}

definition{
  name: floatIntrinsic,
  parameters: [e : Nat, m : Nat, op : String, values : (List Value)],
  result: (Option Nat),
  body: (match values
    | List.nil => (ctor Option.none [Nat])
    | List.cons a rest => (match (floatBitsOf a)
      | Option.none => (ctor Option.none [Nat])
      | Option.some x => (if (prim equal op "fabs" : Bool) then (ctor Option.some [Nat] (Prelude::floatAbs e m x)) else (if (prim equal op "sqrt" : Bool) then (ctor Option.some [Nat] (Prelude::floatSqrt e m x)) else (if (prim equal op "trunc" : Bool) then (ctor Option.some [Nat] (Prelude::floatTrunc e m x)) else (if (prim equal op "floor" : Bool) then (ctor Option.some [Nat] (Prelude::floatFloor e m x)) else (if (prim equal op "ceil" : Bool) then (ctor Option.some [Nat] (Prelude::floatCeil e m x)) else (if (prim equal op "round" : Bool) then (ctor Option.some [Nat] (Prelude::floatRoundHalfAway e m x)) else (if (prim equal op "round_ties_even" : Bool) then (ctor Option.some [Nat] (Prelude::floatRoundTiesEven e m x)) else (match rest
        | List.nil => (ctor Option.none [Nat])
        | List.cons b more => (match (floatBitsOf b)
          | Option.none => (ctor Option.none [Nat])
          | Option.some y => (if (prim equal op "copysign" : Bool) then (ctor Option.some [Nat] (Prelude::floatCopySign e m x y)) else (if (prim equal op "fadd_algebraic" : Bool) then (ctor Option.some [Nat] (Prelude::floatAdd e m x y)) else (if (prim equal op "fsub_algebraic" : Bool) then (ctor Option.some [Nat] (Prelude::floatSub e m x y)) else (if (prim equal op "fmul_algebraic" : Bool) then (ctor Option.some [Nat] (Prelude::floatMul e m x y)) else (if (prim equal op "fdiv_algebraic" : Bool) then (ctor Option.some [Nat] (Prelude::floatDiv e m x y)) else (if (prim equal op "frem_algebraic" : Bool) then (ctor Option.some [Nat] (Prelude::floatRem e m x y)) else (match more
            | List.nil => (ctor Option.none [Nat])
            | List.cons c others => (match (floatBitsOf c)
              | Option.none => (ctor Option.none [Nat])
              | Option.some z => (if (prim equal op "fma" : Bool) then (ctor Option.some [Nat] (Prelude::floatFma e m x y z)) else (ctor Option.none [Nat])))))))))))))))))))))
}

definition{
  name: evalCall,
  parameters: [
    fuel : Nat,
    program : Program,
    frames : (List Frame),
    callee : Callee,
    args : (List Operand),
    destination : Place,
    target : (Option Nat)
  ],
  result: Outcome,
  body: (match callee
    | Callee.body path => (callBody program frames path (nil Generic) args destination target)
    | Callee.instance path generics => (callBody program frames path generics args destination target)
    | Callee.sizeOf generic => (match (genericLayout frames generic)
      | Option.none => (ctor Outcome.undefined "a generic argument whose layout the layout table does not hold")
      | Option.some layout => (finishCall frames destination target (ctor Value.int (project program pointerWidth) false (layoutSize layout))))
    | Callee.operand op => (match (evalOperand program frames op)
      | Evaluated.stop outcome => outcome
      | Evaluated.found value => (match value
        | Value.function path => (callBody program frames path (nil Generic) args destination target)
        | Value.int p q r => (ctor Outcome.undefined "TerminatorKind::Call of a value that is not a function item")
        | Value.boolean p => (ctor Outcome.ub "a call of a value that is not a function")
        | Value.character p => (ctor Outcome.ub "a call of a value that is not a function")
        | Value.unit => (ctor Outcome.ub "a call of a value that is not a function")
        | Value.tuple p => (ctor Outcome.undefined "TerminatorKind::Call of a value that is not a function item")
        | Value.adt p q r => (ctor Outcome.undefined "TerminatorKind::Call of a value that is not a function item")
        | Value.reference r => (ctor Outcome.undefined "TerminatorKind::Call of a value that is not a function item")
        | Value.pointer p q r => (ctor Outcome.undefined "TerminatorKind::Call of a value that is not a function item")
        | Value.text p => (ctor Outcome.ub "a call of a value that is not a function")
        | Value.uninit => (ctor Outcome.ub "a call of an uninitialized value")))
    | Callee.bound name => (match (findBinding (project program dictionary) name)
      | Option.some path => (callBody program frames path (nil Generic) args destination target)
      | Option.none => (ctor Outcome.unlinked name))
    | Callee.intrinsic name => (evalIntrinsic program frames name args destination target)
    | Callee.constant path generics => (match (findBody (project program bodies) path)
      | Option.none => (ctor Outcome.unlinked path)
      | Option.some item => (ctor Outcome.running (cons (enterFrame item (nil Value) destination target true (resolveGenerics frames generics)) frames)))
    | Callee.layoutQuery query generic => (if (prim equal query "align_of" : Bool) then (match (genericAlign frames generic)
      | Option.none => (ctor Outcome.undefined "a generic argument whose alignment the layout table does not hold")
      | Option.some align => (finishCall frames destination target (ctor Value.int (project program pointerWidth) false align))) else (if (prim equal query "assert_inhabited" : Bool) then (match (genericLayout frames generic)
      | Option.none => (ctor Outcome.undefined "a generic argument whose layout the layout table does not hold")
      | Option.some layout => (finishCall frames destination target (ctor Value.unit))) else (ctor Outcome.unlinked query)))
    | Callee.float e m op => (match (evalOperands program frames args)
      | EvaluatedList.stop outcome => outcome
      | EvaluatedList.found values => (match (floatIntrinsic e m op values)
        | Option.none => (ctor Outcome.ub "a float intrinsic of a value that is not a float")
        | Option.some bits => (finishCall frames destination target (floatValue e m bits))))
    | Callee.convert source target2 => (match (evalOperands program frames args)
      | EvaluatedList.stop outcome => outcome
      | EvaluatedList.found values => (match values
        | List.nil => (ctor Outcome.ub "a float_to_int_unchecked of no value")
        | List.cons value others => (match (evalUnchecked source target2 value)
          | Evaluated.stop outcome => outcome
          | Evaluated.found result => (finishCall frames destination target result))))
    | Callee.writeBytes generic => (match (genericLayout frames generic)
      | Option.none => (ctor Outcome.undefined "a generic argument whose layout the layout table does not hold")
      | Option.some element => (match (evalOperands program frames args)
        | EvaluatedList.stop outcome => outcome
        | EvaluatedList.found values => (writeBytes fuel (project program bigEndian) frames element values destination target)))
    | Callee.transmute from to => (match (genericLayout frames from)
      | Option.none => (ctor Outcome.undefined "a generic argument whose layout the layout table does not hold")
      | Option.some source => (match (genericLayout frames to)
      | Option.none => (ctor Outcome.undefined "a generic argument whose layout the layout table does not hold")
      | Option.some target2 => (match (evalOperands program frames args)
      | EvaluatedList.stop outcome => outcome
      | EvaluatedList.found values => (match values
        | List.nil => (ctor Outcome.ub "a transmute of no value")
        | List.cons value others => (match (transmuteValue fuel (project program bigEndian) source target2 value)
          | Evaluated.stop outcome => outcome
          | Evaluated.found result => (finishCall frames destination target result)))))))
}

definition{
  name: evalReturn,
  parameters: [fuel : Nat, frames : (List Frame)],
  result: Outcome,
  body: (match frames
    | List.nil => (ctor Outcome.ub "a return with no frame")
    | List.cons top rest => (match (nthValue (project top locals) 0)
      | Option.none => (ctor Outcome.ub "a return with no return place")
      | Option.some returned => (match (if (project top pinned) then (pinWork fuel (countFrames rest) frames (cons (ctor Pin.visit returned) (nil Pin)) (nil Value)) else (ctor Evaluated.found returned))
      | Evaluated.stop outcome => outcome
      | Evaluated.found result => (match rest
        | List.nil => (ctor Outcome.returned result)
        | List.cons caller below => (match (project top target)
          | Option.none => (ctor Outcome.ub "a return from a call that does not return")
          | Option.some block => (match (placeOf rest (project top destination))
            | Located.stop outcome => outcome
            | Located.found loc => (match (writeAt rest loc result)
              | Option.none => (ctor Outcome.ub "a return into a place that holds no value")
              | Option.some written => (match written
                | List.nil => (ctor Outcome.ub "a return with no frame")
                | List.cons back others => (ctor Outcome.running (cons (moveTo back block) others))))))))))
}

definition{
  name: evalTerminator,
  parameters: [fuel : Nat, program : Program, frames : (List Frame), frame : Frame, terminator : Terminator],
  result: Outcome,
  body: (match terminator
    | Terminator.jump block => (ctor Outcome.running (replaceTop frames (moveTo frame block)))
    | Terminator.switch op arms otherwise => (match (evalOperand program frames op)
      | Evaluated.stop outcome => outcome
      | Evaluated.found value => (match (scalarOf value)
        | Option.none => (ctor Outcome.ub "a switch on a value that is not a scalar")
        | Option.some number => (ctor Outcome.running (replaceTop frames (moveTo frame (switchTarget arms number (reprOfValue value) otherwise))))))
    | Terminator.exit => (evalReturn fuel frames)
    | Terminator.unreachable => (ctor Outcome.ub "an unreachable terminator reached")
    | Terminator.drop block => (ctor Outcome.running (replaceTop frames (moveTo frame block)))
    | Terminator.call destination callee args target => (evalCall fuel program frames callee args destination target)
    | Terminator.assert condition expected kind block => (match (evalOperand program frames condition)
      | Evaluated.stop outcome => outcome
      | Evaluated.found value => (match value
        | Value.boolean b => (if (sameBool b expected) then (ctor Outcome.running (replaceTop frames (moveTo frame block))) else (ctor Outcome.panicked kind))
        | Value.int p q r => (ctor Outcome.ub "an assert on a value that is not a bool")
        | Value.character p => (ctor Outcome.ub "an assert on a value that is not a bool")
        | Value.unit => (ctor Outcome.ub "an assert on a value that is not a bool")
        | Value.tuple p => (ctor Outcome.ub "an assert on a value that is not a bool")
        | Value.adt p q r => (ctor Outcome.ub "an assert on a value that is not a bool")
        | Value.reference r => (ctor Outcome.ub "an assert on a value that is not a bool")
        | Value.pointer p q r => (ctor Outcome.ub "an assert on a value that is not a bool")
        | Value.function p => (ctor Outcome.ub "an assert on a value that is not a bool")
        | Value.text p => (ctor Outcome.ub "an assert on a value that is not a bool")
        | Value.uninit => (ctor Outcome.ub "a read of an uninitialized value")))
    | Terminator.other kind text => (ctor Outcome.undefined kind))
}

definition{
  name: evalStatement,
  parameters: [program : Program, frames : (List Frame), frame : Frame, statement : Statement],
  result: Outcome,
  body: (match statement
    | Statement.assign target value => (evalAssign program frames target value)
    | Statement.marker kind => (ctor Outcome.running (replaceTop frames (advanceFrame frame)))
    | Statement.assume condition => (ctor Outcome.running (replaceTop frames (advanceFrame frame)))
    | Statement.other kind text => (ctor Outcome.undefined kind))
}

definition{
  name: stepFrames,
  parameters: [fuel : Nat, program : Program, frames : (List Frame)],
  result: Outcome,
  body: (match frames
    | List.nil => (ctor Outcome.ub "a step with no frame")
    | List.cons frame rest => (match (nthBlock (project (project frame body) blocks) (project frame block))
      | Option.none => (ctor Outcome.ub "a jump to a block the body does not have")
      | Option.some block => (match (nthStatement (project block statements) (project frame statement))
        | Option.some statement => (evalStatement program frames frame statement)
        | Option.none => (evalTerminator fuel program frames frame (project block terminator)))))
}

definition{
  name: runFrames,
  parameters: [fuel : Nat, program : Program, frames : (List Frame)],
  result: Outcome,
  body: (match fuel
    | Nat.zero => (ctor Outcome.exhausted)
    | Nat.succ rest => (match (stepFrames rest program frames)
      | Outcome.running next => (runFrames rest program next)
      | Outcome.returned value => (ctor Outcome.returned value)
      | Outcome.panicked kind => (ctor Outcome.panicked kind)
      | Outcome.ub reason => (ctor Outcome.ub reason)
      | Outcome.undefined kind => (ctor Outcome.undefined kind)
      | Outcome.unlinked path => (ctor Outcome.unlinked path)
      | Outcome.aborted => (ctor Outcome.aborted)
      | Outcome.exhausted => (ctor Outcome.exhausted))),
  recursive_argument: fuel
}

definition{
  name: evaluate,
  parameters: [fuel : Nat, program : Program, path : String, args : (List Value)],
  result: Outcome,
  body: (match (findBody (project program bodies) path)
    | Option.none => (ctor Outcome.unlinked path)
    | Option.some callee => (runFrames fuel program (cons (enterFrame callee args (record Place {base := 0, projection := (nil Proj)}) (ctor Option.none [Nat]) false (nil Generic)) (nil Frame))))
}
