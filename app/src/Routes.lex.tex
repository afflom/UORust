definition{
  name: agree,
  parameters: [lead : Route, leadTag : String, rest : (List String)],
  result: Verdict,
  body: (match rest
    | List.nil => (ctor Verdict.Routed lead leadTag)
    | List.cons tag more => (if (prim equal tag "" : Bool) then (ctor Verdict.BlankFact) else (match (routeOf tag)
      | Option.none => (ctor Verdict.UnknownFact tag)
      | Option.some route => (if (prim equal (project route id) (project lead id) : Bool) then (agree lead leadTag more) else (ctor Verdict.Ambiguous leadTag tag))))),
  recursive_argument: rest
}

definition{
  name: classify,
  parameters: [tags : (List String)],
  result: Verdict,
  body: (match tags
    | List.nil => (ctor Verdict.Empty)
    | List.cons tag rest => (if (prim equal tag "" : Bool) then (ctor Verdict.BlankFact) else (match (routeOf tag)
      | Option.none => (ctor Verdict.UnknownFact tag)
      | Option.some route => (agree route tag rest))))
}
